    def _get(self, url):
        try:
            r = self.session.get(url, timeout=15)
            r.encoding = 'utf-8'
            if r.status_code == 200 and r.text and len(r.text) > 100:
                return r.text
        except Exception as e:
            self.log(f"_get 请求失败: {url} -> {e}")
        return None

    def _fix(self, u):
        if not u:
            return ""
        if u.startswith("//"):
            return "https:" + u
        if u.startswith("/"):
            return self.host + u
        return u

    def getpq(self, data):
        try:
            return pq(data)
        except Exception:
            try:
                return pq(data.encode('utf-8'))
            except Exception:
                return pq('')

    def _extract_pic_url(self, raw_text):
        if not raw_text:
            return ""
        m = self._re_pic.search(raw_text)
        if m:
            url = m.group(1).strip()
            if url.startswith("http") and not url.endswith(".gif"):
                return url
        return ""

    def _get_pic_proxy(self, raw_text):
        pic_url = self._extract_pic_url(raw_text)
        if pic_url:
            return f"{self.getProxyUrl()}&url={pic_url}&type=img"
        return ""

    def _get_img_from_dom(self, item):
        img = item('img')
        if img:
            pic = img.attr('data-src') or ''
            if not pic or not pic.startswith('http'):
                pic = img.attr('src') or ''
            if pic and pic.startswith('http') and not pic.endswith('.gif'):
                return f"{self.getProxyUrl()}&url={pic}&type=img"
        script_text = item('script').text()
        return self._get_pic_proxy(script_text)

    def aesimg(self, data):
        try:
            cipher = AES.new(self.aes_key, AES.MODE_CBC, self.aes_iv)
            return unpad(cipher.decrypt(data), AES.block_size)
        except Exception:
            return data

    def localProxy(self, param):
        ptype = param.get('type', '')
        if ptype == 'img':
            try:
                url = param.get('url', '')
                if not url or not url.startswith('http'):
                    return [200, 'image/jpeg', b'']
                res = requests.get(url, headers=self.headers, timeout=10)
                if res.status_code == 200 and len(res.content) > 100:
                    return [200, res.headers.get('Content-Type', 'image/jpeg'), self.aesimg(res.content)]
            except Exception:
                pass
            return [200, 'image/jpeg', b'']
        return [200, 'text/plain', b'']

    def _parse_list(self, html):
        if not html:
            return []
        doc = self.getpq(html)
        if not doc:
            return []
        results = []
        seen = set()
        for item in doc('article').items():
            try:
                if item.hasClass('ad-item'):
                    continue
                link = item('a[href*="/archives/"]')
                if not link:
                    continue
                href = link.attr('href') or ""
                if not href or href in seen:
                    continue
                seen.add(href)
                h2 = item('h2.post-card-title')
                title = h2.text().strip() if h2 else ""
                if not title or len(title) < 3:
                    continue
                pic = self._get_img_from_dom(link)
                date_text = ""
                info = item('.post-card-info span')
                if info:
                    for s in info.items():
                        st = s.text()
                        if "年" in st and "日" in st:
                            date_text = st.replace("•", "").strip()
                            break
                results.append({
                    "vod_id": self._fix(href),
                    "vod_name": title,
                    "vod_pic": pic,
                    "vod_remarks": date_text
                })
            except Exception:
                continue
        return results

    def _has_next_page(self, html):
        return 'class="next"' in html or "class='next'" in html

    def _extract_detail_by_regex(self, html):
        title = ""
        m = self._re_h1_titled.search(html)
        if m:
            title = self._re_tag.sub('', m.group(1)).strip()
        if not title:
            m = self._re_h1_any.search(html)
            if m:
                title = self._re_tag.sub('', m.group(1)).strip()

        quality_arr = []
        for match in self._re_config.finditer(html):
            config_str = match.group(1)
            try:
                cfg = json.loads(config_str)
                if "video" in cfg and "url" in cfg["video"]:
                    video_url = cfg["video"]["url"]
                    if video_url:
                        before_config = html[:match.start()]
                        heading_matches = list(self._re_heading.finditer(before_config))
                        name = ""
                        if heading_matches:
                            last_heading = heading_matches[-1].group(1)
                            name = self._re_tag.sub('', last_heading).strip()
                        if (not name or len(name) < 2
                                or any(kw in name for kw in self._skip_kw)
                                or "{{" in name or "}}" in name):
                            name = "视频" + str(len(quality_arr) // 2 + 1)
                        quality_arr.append(name)
                        quality_arr.append(video_url)
            except Exception:
                continue

        vod_play_url = ""
        if quality_arr:
            vod_play_url = "接口免费$" + "$".join(quality_arr)
        else:
            links = self._re_archives_link.findall(html)
            play_items = []
            seen_hrefs = set()
            for href, txt in links:
                txt_clean = self._re_tag.sub('', txt).strip()
                has_jump = "点我查看详情" in txt_clean or "点我跳转" in txt_clean or "点击查看详情" in txt_clean
                if (has_jump
                        and not any(kw in txt_clean for kw in self._skip_kw)
                        and len(txt_clean) > 2):
                    clean_title = self._re_skip_prefix.sub('', txt_clean).strip()
                    clean_title = self._re_skip_suffix1.sub('', clean_title).strip()
                    clean_title = self._re_skip_suffix2.sub('', clean_title).strip()
                    if clean_title and href not in seen_hrefs:
                        seen_hrefs.add(href)
                        clean_href = href.split("#")[0]
                        play_items.append(clean_title + "$" + self._fix(clean_href))
            if play_items:
                vod_play_url = "#".join(play_items)

        return title, vod_play_url

    def playerContent(self, flag, id, vipFlags):
        referer = self.host + "/"
        if "$" in id:
            return {"parse": 0, "url": id.split("$"), "header": {"Referer": referer}}
        if ".m3u8" in id or ".mp4" in id:
            return {"parse": 0, "url": id, "header": {"Referer": referer}}
        if "/archives/" in id:
            url = id if id.startswith("http") else self._fix(id)
            html = self._get(url)
            if html:
                _, vod_play_url = self._extract_detail_by_regex(html)
                if vod_play_url and "未找到视频" not in vod_play_url and "页面加载失败" not in vod_play_url:
                    if vod_play_url.startswith("接口免费$"):
                        rest = vod_play_url[len("接口免费$"):]
                        if "$" in rest:
                            return {"parse": 0, "url": rest.split("$"), "header": {"Referer": referer}}
                        else:
                            return {"parse": 0, "url": rest, "header": {"Referer": referer}}
                    elif "#" in vod_play_url:
                        return {"parse": 1, "url": url, "header": {"Referer": referer}}
            return {"parse": 1, "url": url, "header": {"Referer": referer}}
        return {"parse": 1, "url": id, "header": {"Referer": referer}}
