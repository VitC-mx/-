class _AESCBC:

    @staticmethod
    def encrypt(data, key, iv):
        try:
            from Crypto.Cipher import AES
            return AES.new(key, AES.MODE_CBC, iv).encrypt(_AESCBC.pad(data))
        except Exception:
            from cryptography.hazmat.backends import default_backend
            from cryptography.hazmat.primitives.ciphers import Cipher, algorithms, modes
            enc = Cipher(algorithms.AES(key), modes.CBC(iv), backend=default_backend()).encryptor()
            return enc.update(_AESCBC.pad(data)) + enc.finalize()

    @staticmethod
    def decrypt(data, key, iv):
        try:
            from Crypto.Cipher import AES
            plain = AES.new(key, AES.MODE_CBC, iv).decrypt(data)
        except Exception:
            from cryptography.hazmat.backends import default_backend
            from cryptography.hazmat.primitives.ciphers import Cipher, algorithms, modes
            dec = Cipher(algorithms.AES(key), modes.CBC(iv), backend=default_backend()).decryptor()
            plain = dec.update(data) + dec.finalize()
        return _AESCBC.unpad(plain)

    @staticmethod
    def pad(data):
        n = 16 - len(data) % 16
        return data + bytes([n]) * n

    @staticmethod
    def unpad(data):
        if not data:
            return data
        n = data[-1]
        return data[:-n] if 1 <= n <= 16 else data

    def _api(self, path, data=None, silent=False):
        path = "/" + path.lstrip("/")
        max_attempts = 1 if silent else 2
        for attempt in range(max_attempts):
            rid = str(uuid.uuid4())
            key = self._key(rid)
            iv = os.urandom(16)
            raw = json.dumps(
                {"token": self.token or "", "deviceId": self.device_id, "data": data or {}},
                ensure_ascii=False, separators=(",", ":"),
            ).encode("utf-8")
            body = iv + _AESCBC.encrypt(gzip.compress(raw), key, iv)
            ts = int(time.time())
            sign = hashlib.sha256(
                ("Dart|%s|%s|%s|%s" % (self.session_id, rid, ts, path)).encode("utf-8")
            ).hexdigest() + "-" + str(ts)
            extra = {"time": str(ts), "sign": sign, "requestId": rid}
            try:
                r = self.session.post(self.api + path, data=body, headers=extra, timeout=15)
                r.raise_for_status()
                result = self._decode(r.content, rid)
                if isinstance(result, dict) and result.get("token"):
                    self.token = result["token"]
                return result
            except Exception as e:
                if attempt < max_attempts - 1:
                    continue
                if not silent:
                    self._log("_api %s -> %s" % (path, str(e)[:120]))
                return {}

    def _key(self, rid):
        return hmac.new(
            self.platform_key.encode("utf-8"),
            bytes.fromhex(str(rid).replace("-", "")),
            hashlib.sha256,
        ).digest()

    def _decode(self, blob, rid):
        if not blob:
            return {}
        if len(blob) < 32 or (len(blob) - 16) % 16 != 0:
            try:
                return json.loads(blob.decode("utf-8"))
            except Exception:
                return {}
        try:
            plain = _AESCBC.decrypt(blob[16:], self._key(rid), blob[:16])
            if plain[:2] == b"\x1f\x8b":
                plain = gzip.decompress(plain)
            return json.loads(plain.decode("utf-8"))
        except Exception:
            return {}

    def _nav_filter(self, code):
        if code not in self.filter_cache:
            data = self._api("/drama/navFilter", {"code": str(code)})
            if data:
                self.filter_cache[code] = self._list(data.get("data", data) if isinstance(data, dict) else data)
        return self.filter_cache.get(code, [])

    def _list(self, data, depth=0):
        if isinstance(data, list):
            return data
        if depth > 5 or not isinstance(data, dict):
            return []
        for k in ("list", "items", "data"):
            v = data.get(k)
            if isinstance(v, list):
                return v
            if isinstance(v, dict):
                return self._list(v, depth + 1)
        return []

    def _nav_items(self, data):
        blocks = self._list(data.get("data", data) if isinstance(data, dict) else data)
        items = []
        for b in blocks:
            if not isinstance(b, dict):
                continue
            if isinstance(b.get("items"), list):
                items.extend(b["items"])
            elif b.get("id") or b.get("drama_id"):
                items.append(b)
        return items
