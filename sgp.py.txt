    def _fetch(self, url, headers=None, timeout=15, tries=2):
        hdr = {'User-Agent': self.UA}
        if headers:
            hdr.update(headers)
        last = None
        for _ in range(tries):
            try:
                r = self._sess().get(url, headers=hdr, timeout=timeout)
                r.encoding = 'utf-8'
                return r
            except Exception as e:
                last = e
        raise last

    def _rmk(self, s):
        s = self._tostr(s).strip()
        if not s:
            return ''
        if re.search(r'\d+\s*[:：]\s*\d{2}', s):
            return s
        if '集' in s and re.search(r'\d', s):
            return s
        return ''

    def _join(self, base, url):
        url = self._tostr(url)
        if not url:
            return ''
        if url.startswith('http://') or url.startswith('https://'):
            return url
        if url.startswith('//'):
            return 'https:' + url
        base = base.rstrip('/')
        if url.startswith('/'):
            return base + url
        return base + '/' + url

    def _php_headers(self, site):
        return {'Content-Type': 'application/x-www-form-urlencoded',
                'Origin': site, 'Referer': site + '/'}

    def _php_decrypt(self, s):
        try:
            raw = base64.b64decode(s)
            if not raw or len(raw) % 16 != 0 or AES is None:
                return None
            d = AES.new(self.PHP_KEY, AES.MODE_CBC, self.PHP_IV).decrypt(raw)
            pad = d[-1]
            if 1 <= pad <= 16 and all(b == pad for b in d[-pad:]):
                d = d[:-pad]
            return json.loads(d.decode('utf-8', 'ignore'))
        except Exception:
            return None

    def _php_call(self, api, site, path, body):
        try:
            r = self._post(api + path, data=body, headers=self._php_headers(site), timeout=20)
            j = self._json(r)
            v = j.get('data')
            if isinstance(v, str) and len(v) > 20:
                d = self._php_decrypt(v)
                if d is not None:
                    return d
            return j
        except Exception:
            return {}

    def _aes_encrypt(self, plain, key, iv):
        from Crypto.Cipher import AES
        from Crypto.Util.Padding import pad
        return AES.new(key, AES.MODE_CBC, iv).encrypt(pad(plain, 16))

    def _hd_call(self, path, data):
        last_err = ''
        
        tries = 2 if getattr(self, '_hd_host_idx', None) is not None else 1
        for host in self._hd_hosts():
            try:
                rid = self._hd_uuid()
                device_id = self._hd_uuid()
                key = hmac.new(self.HD_PLATFORM_KEY.encode('utf-8'),
                               bytes.fromhex(rid.replace('-', '')), hashlib.sha256).digest()
                iv = os.urandom(16)
                plain = json.dumps({'token': '', 'deviceId': device_id, 'data': data},
                                   separators=(',', ':')).encode('utf-8')
                cipher = self._aes_encrypt(gzip.compress(plain), key, iv)
                t = str(int(time.time()))
                sign = hashlib.sha256(('Dart|%s|%s|%s|%s' % (device_id, rid, t, path)).encode('utf-8')).hexdigest() + '-' + t
                headers = {
                    'User-Agent': self.HD_UA, 'Accept': '*/*',
                    'Origin': host, 'Referer': host + '/home',
                    'Content-Type': 'application/octet-stream',
                    'version': '2.0.0', 'deviceType': 'web',
                    'requestId': rid, 'sessionId': device_id, 'time': t, 'sign': sign,
                }
                r = self._post(host + '/api' + path, data=iv + cipher, headers=headers, timeout=15, tries=tries)
                self._hd_host_idx = self.HD_HOSTS.index(host)
                blob = r.content
                try:
                    return json.loads(blob)
                except Exception:
                    pass
                if len(blob) < 32:
                    raise Exception('response too short')
                from Crypto.Cipher import AES
                from Crypto.Util.Padding import unpad
                plain2 = unpad(AES.new(key, AES.MODE_CBC, blob[:16]).decrypt(blob[16:]), 16)
                if plain2[:2] == b'\x1f\x8b':
                    plain2 = gzip.decompress(plain2)
                return json.loads(plain2)
            except Exception as e:
                last_err = str(e)
        raise Exception(last_err or 'huangdou failed')

    def _hd_pool(self, need, max_new=3):
        now = int(time.time())
        p = getattr(self, '_hd_pool_v', None)
        if p is None or now - p.get('_ts', 0) > 1800:
            p = {'_items': [], '_ids': set(), '_pages': 0, '_done': False, '_ts': now}
            self._hd_pool_v = p
        grown = 0
        while not p['_done'] and len(p['_items']) < need and grown < max_new:
            p['_pages'] += 1
            try:
                recs = self._hd_list(p['_pages'], 1000)
            except Exception:
                recs = []
            if not recs:
                p['_done'] = True
                break
            for it in recs:
                did = self._tostr(it.get('id') or it.get('drama_id'))
                if not did or did in p['_ids']:
                    continue
                p['_ids'].add(did)
                item = self._hd_item(it)
                if item:
                    item['_cat'] = self._tostr(it.get('category'))
                    p['_items'].append(item)
            grown += 1
            if len(recs) < 1000:
                p['_done'] = True
                break
        return p

    def _proxyPic(self, params):
        u = self._tostr(params.get('url'))
        if not u.startswith('http'):
            return [404, 'text/plain; charset=utf-8', 'bad url']
        try:
            r = requests.get(u, headers={'User-Agent': self.UA, 'Referer': '*',
                                         'Accept': 'image/*'}, timeout=15, verify=False)
            data = r.content
            if not data:
                return [404, 'text/plain; charset=utf-8', 'empty']
            img = self._decryptImg(data)
            if not img:
                return [404, 'text/plain; charset=utf-8', 'decrypt fail']
            mime = self._imgMime(img)
            return [200, mime, img]
        except Exception as e:
            self._log('proxyPic err: %s' % e)
            return [404, 'text/plain; charset=utf-8', 'err']

    @staticmethod
    def _isPlainImg(data):
        return (data[:3] == b'\xff\xd8\xff' or data[:8] == b'\x89PNG\r\n\x1a\n'
                or data[:6] in (b'GIF87a', b'GIF89a')
                or (data[:4] == b'RIFF' and data[8:12] == b'WEBP'))

    def _decryptImg(self, data):
        if not data:
            return b''
        if self._isPlainImg(data):
            return data
        if AES is None or len(data) % 16 != 0:
            return b''
        try:
            dec = AES.new(self.MEDIA_KEY, AES.MODE_CBC, self.MEDIA_IV).decrypt(data)
            pad = dec[-1]
            if 1 <= pad <= 16 and all(b == pad for b in dec[-pad:]):
                dec = dec[:-pad]
            else:
                dec = dec.rstrip(b'\x00')
            if self._isPlainImg(dec):
                return dec
            return dec
        except Exception:
            return b''

    @staticmethod
    def _imgMime(data):
        if data[:3] == b'\xff\xd8\xff':
            return 'image/jpeg'
        if data[:8] == b'\x89PNG\r\n\x1a\n':
            return 'image/png'
        if data[:6] in (b'GIF87a', b'GIF89a'):
            return 'image/gif'
        if data[:4] == b'RIFF' and data[8:12] == b'WEBP':
            return 'image/webp'
        return 'image/jpeg'
