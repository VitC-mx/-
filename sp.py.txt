class _RParser(object):

    def __init__(self, text):
        self.s = text
        self.n = len(text)
        self.pos = 0
        self.R = {}

    def _skip(self):
        s = self.s
        while self.pos < self.n and s[self.pos] in ' \t\r\n':
            self.pos += 1

    def parse(self):
        self._skip()
        return self._value()

    def _value(self):
        self._skip()
        if self.pos >= self.n:
            return None
        c = self.s[self.pos]
        if c == '{':
            return self._object()
        if c == '[':
            return self._array()
        if c in '"\'':
            return self._string()
        if c == '$':
            return self._ref()
        return self._literal()

    def _object(self):
        self.pos += 1
        obj = {}
        while True:
            self._skip()
            if self.pos >= self.n:
                return obj
            c = self.s[self.pos]
            if c == '}':
                self.pos += 1
                return obj
            if c in '"\'':
                key = self._string()
            else:
                start = self.pos
                while self.pos < self.n and (self.s[self.pos].isalnum() or self.s[self.pos] in '_$'):
                    self.pos += 1
                key = self.s[start:self.pos]
                if not key:
                    self.pos += 1
                    continue
            self._skip()
            if self.pos < self.n and self.s[self.pos] == ':':
                self.pos += 1
            obj[key] = self._value()
            self._skip()
            if self.pos < self.n and self.s[self.pos] == ',':
                self.pos += 1

    def _array(self):
        self.pos += 1
        arr = []
        while True:
            self._skip()
            if self.pos >= self.n:
                return arr
            if self.s[self.pos] == ']':
                self.pos += 1
                return arr
            arr.append(self._value())
            self._skip()
            if self.pos < self.n and self.s[self.pos] == ',':
                self.pos += 1

    _ESC = {'"': '"', "'": "'", '\\': '\\', '/': '/', 'b': '\b', 'f': '\f', 'n': '\n', 'r': '\r', 't': '\t'}

    def _string(self):
        q = self.s[self.pos]
        self.pos += 1
        s = self.s
        out = []
        while self.pos < self.n:
            c = s[self.pos]
            if c == q:
                self.pos += 1
                return ''.join(out)
            if c == '\\':
                self.pos += 1
                if self.pos >= self.n:
                    break
                e = s[self.pos]
                if e == 'u':
                    hexs = s[self.pos + 1:self.pos + 5]
                    try:
                        out.append(chr(int(hexs, 16)))
                    except Exception:
                        out.append(hexs)
                    self.pos += 5
                    continue
                if e == 'x':
                    hexs = s[self.pos + 1:self.pos + 3]
                    try:
                        out.append(chr(int(hexs, 16)))
                    except Exception:
                        out.append(hexs)
                    self.pos += 3
                    continue
                out.append(self._ESC.get(e, e))
                self.pos += 1
                continue
            out.append(c)
            self.pos += 1
        return ''.join(out)

    def _ref(self):
        if self.s[self.pos:self.pos + 2] != '$R':
            return self._literal()
        self.pos += 2
        self._skip()
        if self.pos >= self.n or self.s[self.pos] != '[':
            return None
        self.pos += 1
        start = self.pos
        while self.pos < self.n and self.s[self.pos].isdigit():
            self.pos += 1
        try:
            idx = int(self.s[start:self.pos])
        except Exception:
            return None
        if self.pos < self.n and self.s[self.pos] == ']':
            self.pos += 1
        self._skip()
        if self.pos < self.n and self.s[self.pos] == '=' and self.s[self.pos + 1:self.pos + 2] != '=':
            self.pos += 1
            value = self._value()
            self.R[idx] = value
            return value
        return self.R.get(idx)

    def _literal(self):
        s = self.s
        c = s[self.pos]
        if c == '!':
            flag = s[self.pos + 1:self.pos + 2]
            self.pos += 2
            return flag != '0'
        if c.isdigit() or c in '-.':
            start = self.pos
            while self.pos < self.n and (s[self.pos].isalnum() or s[self.pos] in '.+-'):
                self.pos += 1
            raw = s[start:self.pos]
            try:
                return int(raw)
            except Exception:
                pass
            try:
                return float(raw)
            except Exception:
                return raw
        start = self.pos
        while self.pos < self.n and (s[self.pos].isalpha() or s[self.pos] == '_'):
            self.pos += 1
        word = s[start:self.pos]
        if word in ('null', 'undefined'):
            return None
        if word == 'true':
            return True
        if word == 'false':
            return False
        return word

    def _request(self, url, params=None, timeout=20):
        headers = self._headers()
        try:
            return self.fetch(url, params=params, headers=headers, timeout=timeout, verify=False)
        except Exception:
            pass
        try:
            return requests.get(url, params=params, headers=headers, timeout=timeout,
                                verify=False, proxies=self._proxies())
        except Exception:
            return None

    def _parse_ssr(self, html):
        parts = re.findall(r'<script[^>]*data-tsr-stream-part[^>]*>([\s\S]*?)</script>', html)
        router_text = ''
        for part in parts:
            if '$_TSR.router=' in part:
                router_text = part
                break
        if not router_text:
            return {}
        m = re.search(r'\$_TSR\.router=\s*\([^)]*=>', router_text)
        if not m:
            return {}
        brace = router_text.find('{', m.end())
        if brace < 0:
            return {}
        try:
            parser = _RParser(router_text)
            parser.pos = brace
            router = parser.parse()
        except Exception as exc:
            self._log('ssr parse failed: %s' % exc)
            return {}
        if not isinstance(router, dict):
            return {}
        matches = router.get('matches')
        if not isinstance(matches, list):
            return {}
        for item in reversed(matches):
            if isinstance(item, dict) and isinstance(item.get('l'), dict):
                return item['l']
        return {}

    def _loader(self, path, params=None):
        try:
            url = self.HOST + quote(path, safe='/')
        except Exception:
            return {}
        resp = self._request(url, params)
        if resp is None:
            return {}
        text = getattr(resp, 'text', '') or ''
        if not text:
            return {}
        return self._parse_ssr(text)

    def _total_page(self, loader, page, count):

                tp = int(tp)
            except Exception:
                tp = 0
            if tp > 0:
                return max(page, tp)
        if count <= 0:
            return page
        return page + 1

    def _video_list(self, entries, mode='category'):
        out = []
        if not isinstance(entries, list):
            return out
        for entry in entries:
            item = self._video_item(entry, mode)
            if item:
                out.append(item)
        return out

    def _video_item(self, entry, mode='category'):
        if not isinstance(entry, dict):
            return None
        vid = self._safe_text(entry.get('id'))
        if not vid:
            return None
        name = self._safe_text(entry.get('nameZh')) or self._safe_text(entry.get('name')) or \
            self._safe_text(entry.get('title'))
        pic = self._safe_text(entry.get('coverImageUrl'))
        code = self._safe_text(entry.get('vid'))
        code = code if self._is_vid_code(code) else ''
        duration = self._duration_text(entry.get('duration'))
        if mode == 'search':
            remarks = duration + ('|' + code if code else '')
            year = ''
        else:
            remarks = duration
            year = code
        return {
            'vod_id': vid,
            'vod_name': name,
            'vod_pic': pic,
            'vod_remarks': remarks,
            'vod_year': year,
            'vod_tag': '',
        }

    def _series_list(self, entries):
        out = []
        if not isinstance(entries, list):
            return out
        for entry in entries:
            item = self._series_item(entry)
            if item:
                out.append(item)
        return out

    def _series_item(self, entry):
        if not isinstance(entry, dict):
            return None
        sid = self._safe_text(entry.get('id')) or self._safe_text(entry.get('firstEpisodeId'))
        if not sid:
            return None
        name = self._safe_text(entry.get('nameZh')) or self._safe_text(entry.get('name'))
        pic = self._safe_text(entry.get('coverImageUrl'))
        return {
            'vod_id': 's:' + sid,
            'vod_name': name,
            'vod_pic': pic,
            'vod_remarks': self._episode_count_text(entry),
            'vod_year': '',
            'vod_tag': '',
        }

    def _episode_count_text(self, entry):
        if not isinstance(entry, dict):
            return ''
        label = self._safe_text(entry.get('epLabel'))
        if label:
            return label
        count = entry.get('episodeCount')
        if count is None:
            eps = entry.get('episodes')
            count = len(eps) if isinstance(eps, list) else 0
        try:
            count = int(count)
        except Exception:
            count = 0
        if count > 0:
            return '%d集' % count
        return self._duration_text(entry.get('duration'))

    def _tags_content(self, tags_original, tags_zh, vid_code):
        parts = []
        tags_original = tags_original if isinstance(tags_original, list) else []
        tags_zh = tags_zh if isinstance(tags_zh, list) else []
        for index, original in enumerate(tags_original):
            original_text = self._safe_text(original)
            if not original_text:
                continue
            display = ''
            if index < len(tags_zh) and self._safe_text(tags_zh[index]):
                display = self._safe_text(tags_zh[index])
            else:
                display = original_text
            parts.append(self._click_item('/t/' + original_text, display))
        code = self._safe_text(vid_code)
        if code and self._is_vid_code(code):
            parts.append(self._click_item('search:' + code, code))
        if not parts:
            return ''
        return '[标签] ' + '，'.join(parts)

    def _click_item(self, target, name):
        payload = json.dumps({'id': target, 'name': self._safe_text(name)},
                             ensure_ascii=False, separators=(',', ':'))
        return '[a=cr:%s/]%s[/a]' % (payload, self._safe_text(name))

    def _duration_text(self, seconds):
        try:
            seconds = int(round(float(seconds)))
        except Exception:
            return ''
        seconds = max(0, seconds)
        if seconds <= 0:
            return ''
        hours = seconds // 3600
        minutes = (seconds % 3600) // 60
        secs = seconds % 60
        if hours > 0:
            return '%d小时%02d分%02d秒' % (hours, minutes, secs)
        if minutes > 0:
            return '%d分%02d秒' % (minutes, secs)
        return '%d秒' % secs

    def _is_vid_code(self, value):
        s = self._safe_text(value)
        if not s or len(s) > 24:
            return False
        if re.search(r'[一-鿿]', s):
            return False
        if not re.search(r'[A-Za-z]', s) or not re.search(r'\d', s):
            return False
        return True

    def normalize_id(self, value):
        vid = value[0] if isinstance(value, list) and value else value
        vid = str(vid or '').strip()
        if vid.startswith(self.HOST):
            vid = vid[len(self.HOST):]
        if vid.startswith('/v/'):
            vid = vid[3:]
        vid = vid.split('?')[0].split('#')[0].strip('/')
        allowed = set('0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz_-')
        return ''.join(ch for ch in vid if ch in allowed)

    def hls_api_url(self, vid):
        return self.HOST + '/api/hls/' + vid

    def fetch_index(self, vid, rng=None):
        head = self._headers()
        if rng:
            head['Range'] = rng
        try:
            r = self.fetch(self.hls_api_url(vid), headers=head, timeout=25, verify=False)
        except Exception:
            r = None
        if r is None:
            try:
                r = requests.get(self.hls_api_url(vid), headers=head, timeout=25,
                                 verify=False, proxies=self._proxies())
            except Exception:
                return None
        if r is None:
            return None
        if getattr(r, 'status_code', 0) == 404:
            return None
        return r

    def fetch_segment(self, url):
        head = self._headers()
        try:
            r = self.fetch(url, headers=head, timeout=30, verify=False)
        except Exception:
            r = None
        if r is None:
            try:
                r = requests.get(url, headers=head, timeout=30, verify=False, proxies=self._proxies())
            except Exception:
                return None
        if r is None:
            return None
        return self.unwrap(getattr(r, 'content', None) or b'')

    def unwrap(self, data):
        if not data or data[:8] != self.PNG_SIG:
            return data
        payload = b''
        pos = 8
        size = len(data)
        while pos + 8 <= size:
            length = int.from_bytes(data[pos:pos + 4], 'big')
            name = data[pos + 4:pos + 8]
            if name == b'roUd':
                payload = data[pos + 8:pos + 8 + length]
                break
            if name == b'IEND':
                break
            pos += 12 + length
        if not payload:
            return data
        body = payload[1:]
        if body[:1] == b'G':
            return body
        for wbits in (15, -15, 47):
            try:
                return zlib.decompress(body, wbits)
            except Exception:
                continue
        return body

    def has_seg(self, entry, idx):
        return bool(entry) and 0 <= idx < len(entry.get('segs') or [])

    def parse_range(self, value, total):
        text = str(value or '').strip().lower()
        if total <= 0 or not text.startswith('bytes='):
            return None, None
        span = text[6:].split(',')[0].strip()
        first, _, last = span.partition('-')
        if not first:
            length = self._to_int(last, 0)
            if length <= 0:
                return None, None
            start = max(total - length, 0)
            end = total - 1
        else:
            start = self._to_int(first, -1)
            end = self._to_int(last, total - 1) if last else total - 1
        if start < 0 or start >= total:
            return None, None
        end = min(end, total - 1)
        if end < start:
            return None, None
        return start, end

    def _to_int(self, value, default=0):
        try:
            return int(float(str(value).strip()))
        except Exception:
            return default

    def _build_playlist(self, final_url, content):
        text = self.unwrap(content).decode('utf-8', 'ignore')
        if '#EXTM3U' not in text:
            return None
        segs = []
        lines = []
        for line in text.split('\n'):
            line = line.strip()
            if not line:
                continue
            if line.startswith('#'):
                lines.append(line)
                continue
            lines.append(len(segs))
            segs.append(urljoin(final_url, line))
        if not segs:
            return None
        return {'segs': segs, 'lines': lines, 'expiresAt': int(time.time()) + self.PLAY_TTL}

    def _store_playlist(self, vid, entry):
        cache = dict(self.playlists)
        cache[vid] = entry
        if len(cache) > 3:
            stale = sorted(cache, key=lambda k: cache[k].get('expiresAt', 0))
            for key in stale[:len(cache) - 3]:
                cache.pop(key, None)
        self.playlists = cache

    def resolve_playlist(self, vid, refresh=False):
        now = int(time.time())
        entry = self.playlists.get(vid)
        if entry and not refresh and entry.get('expiresAt', 0) > now:
            return entry
        r = self.fetch_index(vid)
        if r is None:
            return entry
        final_url = getattr(r, 'url', '') or self.hls_api_url(vid)
        entry = self._build_playlist(final_url, getattr(r, 'content', None) or b'')
        if not entry:
            self._log('playlist parse failed: %s' % vid)
            return self.playlists.get(vid)
        self._store_playlist(vid, entry)
        return entry

    def render_playlist(self, vid, entry):
        prefix = 'http://127.0.0.1:9978/proxy?do=py&type=ts&id=%s&i=' % quote(vid, safe='')
        out = []
        for item in entry['lines']:
            if isinstance(item, int):
                out.append(prefix + str(item))
            else:
                out.append(item)
        return '\n'.join(out) + '\n'

    def proxy_playlist(self, vid):
        entry = self.resolve_playlist(vid)
        if not entry:
            entry = self.resolve_playlist(vid, True)
        if not entry:
            return [404, 'text/plain', '', {}]
        return [200, self.M3U8_MIME, self.render_playlist(vid, entry)]

    def proxy_segment(self, vid, index, range_header):
        idx = self._to_int(index, -1)
        entry = self.resolve_playlist(vid)
        if not self.has_seg(entry, idx):
            entry = self.resolve_playlist(vid, True)
        if not self.has_seg(entry, idx):
            return [404, 'text/plain', '', {}]
        data = self.fetch_segment(entry['segs'][idx])
        if data is None:
            entry = self.resolve_playlist(vid, True)
            if not self.has_seg(entry, idx):
                return [404, 'text/plain', '', {}]
            data = self.fetch_segment(entry['segs'][idx])
        if data is None:
            return [404, 'text/plain', '', {}]
        total = len(data)
        start, end = self.parse_range(range_header, total)
        if start is None:
            return [206, self.TS_MIME, data, {'Accept-Ranges': 'bytes'}]
        return [206, self.TS_MIME, data[start:end + 1],
                {'Accept-Ranges': 'bytes', 'Content-Range': 'bytes %d-%d/%d' % (start, end, total)}]

    def localProxy(self, params):
        if not isinstance(params, dict):
            return [404, 'text/plain', '', {}]
        ptype = params.get('type')
        if ptype == 'm3u8':
            return self.proxy_playlist(self.normalize_id(params.get('id')))
        if ptype == 'ts':
            return self.proxy_segment(self.normalize_id(params.get('id')),
                                     params.get('i'), params.get('range'))
        return [404, 'text/plain', '', {}]

    def derive_direct(self, text, final_url):
        for line in text.split('\n')[:-1]:
            line = line.strip()
            if not line or line.startswith('#'):
                continue
            seg = urljoin(final_url, line)
            path, sep, query = seg.partition('?')
            if path.count('/') < 4:
                return ''
            guess = path.rsplit('/', 1)[0] + '/index.m3u8' + (sep + query)
            try:
                r = self.fetch(guess, headers=self._headers(), timeout=15, verify=False)
                if r is not None and (getattr(r, 'content', b'') or b'')[:7] == b'#EXTM3U':
                    return guess
            except Exception:
                pass
            return ''
        return ''

    def playerContent(self, flag, id, vipFlags):
        vid = self.normalize_id(id)
        if not vid:
            return {'parse': 0, 'url': '', 'msg': 'no video id'}
        try:
            r = self.fetch_index(vid, 'bytes=0-4095')
            if r is not None:
                content = getattr(r, 'content', b'') or b''
                final_url = getattr(r, 'url', '') or self.hls_api_url(vid)
                if content[:7] == b'#EXTM3U':
                    direct = self.derive_direct(content.decode('utf-8', 'ignore'), final_url)
                    if direct:
                        return {'parse': 0, 'url': direct, 'header': self._headers(), 'playUrl': ''}
                else:
                    entry = self._build_playlist(final_url, content)
                    if entry:
                        self._store_playlist(vid, entry)
        except Exception:
            pass
        proxy = 'http://127.0.0.1:9978/proxy?do=py&type=m3u8&id=' + quote(vid, safe='')
        return {'parse': 0, 'url': proxy, 'header': self._headers(), 'playUrl': ''}
