    def _discover_online_url(self):
        try:
            nav_domains = self._get_nav_domains()
            if not nav_domains:
                return self.default_base_url
            for domain in nav_domains:
                online_url = self._get_online_url_from_nav(domain)
                if online_url:
                    return online_url
            return self.default_base_url
        except:
            return self.default_base_url

    def _get_nav_domains(self):
        content = self._fetch_dtxt()
        if not content:
            return []
        return self._decrypt_dtxt(content)

    def _fetch_dtxt(self):
        for source in self.dtxt_sources:
            try:
                resp = requests.get(source, timeout=10, headers={'User-Agent': self.ua})
                if resp.status_code == 200 and resp.text.strip():
                    return resp.text.strip()
            except:
                continue
        return None

    def _decrypt_dtxt(self, content):
        try:
            key = bytes.fromhex(self.dtxt_key_hex)
            raw = base64.b64decode(content)
            nonce = raw[:12]
            ciphertext = raw[12:-16]
            tag = raw[-16:]
            cipher = AES.new(key, AES.MODE_GCM, nonce=nonce)
            decrypted = cipher.decrypt_and_verify(ciphertext, tag)
            data = json.loads(decrypted.decode('utf-8'))
            return data.get('dl', [])
        except:
            return []

    def _get_online_url_from_nav(self, nav_domain):
        try:
            resp = requests.get(f"{nav_domain}/api/site-config", timeout=10,
                               headers={'User-Agent': self.ua, 'Accept': 'application/json'})
            if resp.status_code != 200:
                return None
            data = resp.json()
            entries = data.get('data', {}).get('releasePage', {}).get('entries', [])
            for entry in entries:
                title = entry.get('title', '')
                url = entry.get('url', '')
                if '在线H5观看' in title and url:

            return None
        except:
            return None

    def _derive_aes_key(self, nonce):
        nonce_no_dash = nonce.replace("-", "")
        nonce_bytes = bytes.fromhex(nonce_no_dash)
        key_bytes = self.platform_key.encode('utf-8')
        return hmac.new(key_bytes, nonce_bytes, hashlib.sha256).digest()

    def _encrypt_body(self, data, nonce):
        json_str = json.dumps(data, separators=(',', ':'), ensure_ascii=False)
        utf8_bytes = json_str.encode('utf-8')
        compressed = gzip.compress(utf8_bytes)
        aes_key = self._derive_aes_key(nonce)
        iv = os.urandom(16)
        cipher = AES.new(aes_key, AES.MODE_CBC, iv)
        ciphertext = cipher.encrypt(pad(compressed, AES.block_size))
        return iv + ciphertext

    def _decrypt_response(self, body, nonce):
        if len(body) == 0:
            return None
        try:
            text = body.decode('utf-8')
            return json.loads(text)
        except:
            pass
        try:
            aes_key = self._derive_aes_key(nonce)
            if len(body) < 32 or len(body) % 16 != 0:
                return None
            iv = body[:16]
            ciphertext = body[16:]
            cipher = AES.new(aes_key, AES.MODE_CBC, iv)
            decrypted = unpad(cipher.decrypt(ciphertext), AES.block_size)
            if len(decrypted) >= 2 and decrypted[0] == 0x1f and decrypted[1] == 0x8b:
                decompressed = gzip.decompress(decrypted)
                return json.loads(decompressed.decode('utf-8'))
            else:
                return json.loads(decrypted.decode('utf-8'))
        except:
            return None

    def _make_sign(self, nonce, timestamp, url):
        url_without_protocol = url.replace("https://", "").replace("http://", "")
        sign_str = f"Dart|{self.session_id}|{nonce}|{timestamp}|{url_without_protocol}"
        md5_hash = hashlib.md5(sign_str.encode('utf-8')).hexdigest()
        return f"{md5_hash}-{timestamp}"

    def _make_body(self, route, api_params=None):
        data = {"route": route, "params": None}
        if api_params:
            data.update(api_params)
        return {"token": self.token, "deviceId": self.device_id, "data": data}

            headers = {

                'version': '1.0.0',
                'deviceType': 'web',
                'time': timestamp,
                'sign': sign,

            resp = self.session.post(full_url, data=body, headers=headers, timeout=30)
